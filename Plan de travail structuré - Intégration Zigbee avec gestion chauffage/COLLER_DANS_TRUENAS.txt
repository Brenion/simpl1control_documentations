#!/bin/bash
U="http://192.168.0.167:30034"
R=$(curl -s -X POST "$U/api/access-tokens" -H "Content-Type: application/json" -d '{"emailOrUsername":"patrickpagnoulle@gmail.com","password":"6i.v4W92NHi4gFLs"}')
T=$(echo "$R"|grep -o '"item":"[^"]*"'|head -1|sed 's/"item":"//;s/"//')
if [ -z "$T" ]; then echo "ERREUR AUTH: $R"; exit 1; fi
echo "Token OK"
H1="Authorization: Bearer $T"
H2="Content-Type: application/json"
xid(){ echo "$1"|grep -o '"id":"[^"]*"'|head -1|sed 's/"id":"//;s/"//'; }
P=$(curl -s -X POST "$U/api/projects" -H "$H1" -H "$H2" -d '{"name":"Smart Control - Integration Zigbee et Chauffage"}')
PID=$(xid "$P")
echo "Projet: $PID"
mkb(){ xid "$(curl -s -X POST "$U/api/projects/$PID/boards" -H "$H1" -H "$H2" -d "{\"name\":\"$1\",\"position\":$2}")"; }
mkl(){ xid "$(curl -s -X POST "$U/api/boards/$1/lists" -H "$H1" -H "$H2" -d "{\"name\":\"$2\",\"position\":$3}")"; }
mkc(){ curl -s -X POST "$U/api/lists/$1/cards" -H "$H1" -H "$H2" -d "{\"name\":\"$2\",\"position\":$3}" >/dev/null; }
N=0
phase(){
local BID=$(mkb "$1" $2)
echo "Board: $1 -> $BID"
local sec="" LID="" sn=0 cn=0
while IFS='|' read -r s t; do
if [ "$s" != "$sec" ]; then sec="$s"; LID=$(mkl "$BID" "$s" $((sn*65535))); sn=$((sn+1)); cn=0; echo "  Liste: $s"; fi
mkc "$LID" "$t" $((cn*65535)); cn=$((cn+1)); N=$((N+1))
[ $((N%20)) -eq 0 ] && echo "  ... $N cartes"
sleep 0.1
done
}
